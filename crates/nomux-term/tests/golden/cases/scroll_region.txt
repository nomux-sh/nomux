# The scroll region (DECSTBM) and the origin mode (DECOM).

case DECSTBM with the top margin at the bottom margin is ignored
follows DEC STD 070 DECSTBM: the top margin must be less than the bottom margin; xterm ctlseqs "CSI Ps ; Ps r"
input \e[2;3H\e[3;3rX
screen
|                    |
|  X                 |
|                    |
|                    |
|                    |
cursor 1,3

case DECSTBM with a bottom margin after the last row stops at the last row
follows xterm ctlseqs "CSI Ps ; Ps r": the default and the largest bottom margin is the last row
input one\r\ntwo\e[2;99r\e[5H\n
screen
|one                 |
|                    |
|                    |
|                    |
|                    |
cursor 4,0

case setting DECOM moves the cursor to the first cell of the region
follows DEC STD 070 DECOM; vttest "origin mode test"
input \e[2;4r\e[3;3H\e[?6hX
screen
|                    |
|X                   |
|                    |
|                    |
|                    |
cursor 1,1
modes +ORIGIN

case resetting DECOM moves the cursor to the first cell of the screen
follows DEC STD 070 DECOM
input \e[2;4r\e[?6h\e[2;2H\e[?6lX
gap Resetting DECOM keeps the cursor
screen
|X                   |
|                    |
|                    |
|                    |
|                    |
cursor 0,1
library
|                    |
|                    |
| X                  |
|                    |
|                    |
cursor 2,2

case IL outside the scroll region moves no row
follows DEC STD 070 IL: no effect outside the margins; xterm ctlseqs "CSI Ps L"
input one\r\ntwo\r\nthree\e[1;2r\e[3H\e[L
screen
|one                 |
|two                 |
|three               |
|                    |
|                    |
cursor 2,0

case DL outside the scroll region moves no row
follows DEC STD 070 DL: no effect outside the margins; xterm ctlseqs "CSI Ps M"
input one\r\ntwo\r\nthree\e[1;2r\e[3H\e[M
screen
|one                 |
|two                 |
|three               |
|                    |
|                    |
cursor 2,0

case RI in the first row above the scroll region scrolls nothing
follows DEC STD 070 RI: the screen scrolls only at the top margin
input one\r\ntwo\e[2;4r\e[H\eM
screen
|one                 |
|two                 |
|                    |
|                    |
|                    |
cursor 0,0

case SU of more rows than the region has empties the region
follows ECMA-48 8.3.147 SU; xterm ctlseqs "CSI Ps S"
input one\r\ntwo\r\nthree\r\nfour\e[2;3r\e[9S
screen
|one                 |
|                    |
|                    |
|four                |
|                    |
cursor 0,0

case a wrap in the last row of the region scrolls the region only
follows DEC STD 070 autowrap: the wrap is a line feed at the bottom margin
input top\r\n\r\n\r\nfour\e[2;3r\e[3HabcdefghijklmnopqrstXY
screen
|top                 |
|abcdefghijklmnopqrst|
|XY                  |
|four                |
|                    |
cursor 2,2
