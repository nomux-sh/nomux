# Erase variants that full-screen programs send.

case ED 3 keeps the visible screen
follows xterm ctlseqs "CSI 3 J": erase the saved lines only
input one\r\ntwo\e[3J
screen
|one                 |
|two                 |
|                    |
|                    |
|                    |
cursor 1,3

case DECSEL erases the row of the cursor
follows DEC STD 070 DECSEL; xterm ctlseqs "CSI ? Ps K": without DECSCA protection it erases as EL
input abc\r\ndef\e[1;2H\e[?2K
gap DECSEL is not handled: the row keeps its text
screen
|                    |
|def                 |
|                    |
|                    |
|                    |
cursor 0,1
library
|abc                 |
|def                 |
|                    |
|                    |
|                    |
cursor 0,1

case DECSED erases the screen
follows DEC STD 070 DECSED; xterm ctlseqs "CSI ? Ps J": without DECSCA protection it erases as ED
input abc\r\ndef\e[1;2H\e[?2J
gap DECSED is not handled: the screen keeps its text

case DECALN fills the screen with E and moves the cursor to the first cell
follows DEC STD 070 DECALN; vttest "screen alignment"
input abc\e[2;4r\e[3;3H\e#8
